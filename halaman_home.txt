class HomePage extends StatefulWidget{

  @override
  State <HomePage> createState() => _HomePageState();


}

class _HomePageState extends State <HomePage>{
  int currentIndex = 0;

  final List<Widget>pages = const [
    HomeContent(),
    PetaPage(),
    SmartphonePage(),
    Headphone(),
    SmartwatchPage(),
  ];

  @override
  Widget build(BuildContext context){
    return Scaffold(
      appBar: AppBar(title: Text('Toko Gadget'),
      ),
      body: pages[currentIndex],
      bottomNavigationBar: NavigationBar(selectedIndex: currentIndex,onDestinationSelected: (index) {
        setState(() {
          currentIndex = index;
        });
      },
      destinations: const [
        NavigationDestination(
          icon: Icon(Icons.home),
          label: 'home',
        ),
        NavigationDestination(
          icon: Icon(Icons.map), 
          label: 'peta',
          ),
        NavigationDestination(
          icon: Icon(Icons.smartphone), 
          label: 'HP',
          ),
        NavigationDestination(
          icon: Icon(Icons.headphones), 
          label: 'Headphone',
          ),
        NavigationDestination(
          icon: Icon(Icons.watch), 
          label: 'smartwatch',
          ),
      ],
      ),
      );
  }
}
